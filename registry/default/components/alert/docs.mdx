# Alert

An accessible callout banner for warnings, errors, and system announcements.

## Usage

```tsx
import { Alert } from "@/components/ui/Alert";

export default function Example() {
  return (
    <Alert
      variant="destructive"
      title="Connection Terminated"
      description="Your mobile session timed out. Please reconnect."
    />
  );
}
```
